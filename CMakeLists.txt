cmake_minimum_required(VERSION 3.18)
project(lowpoly CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(LOWPOLY_CUDA "build the CUDA backend (Linux/Windows)" OFF)

set(SRC lowpoly.cpp)
set(LIBS)
set(DEFS)
set(GPU_ON OFF)
if(APPLE)
  enable_language(OBJCXX)
  list(APPEND SRC metal.mm)
  list(APPEND LIBS "-framework Metal" "-framework Foundation" "-framework ImageIO" "-framework CoreGraphics" "-framework CoreServices")
  set(GPU_ON ON)
endif()
if(LOWPOLY_CUDA AND NOT APPLE)
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 17)
  list(APPEND SRC cuda.cu)
  set(GPU_ON ON)
endif()
if(NOT GPU_ON)
  list(APPEND DEFS LOWPOLY_NO_GPU)
endif()
if(WIN32)
  list(APPEND LIBS shell32)
else()
  find_package(Threads REQUIRED)
  list(APPEND LIBS Threads::Threads)
endif()

foreach(t lowpoly lowpoly-test)
  add_executable(${t} ${SRC})
  target_compile_definitions(${t} PRIVATE ${DEFS})
  target_link_libraries(${t} PRIVATE ${LIBS})
  if(MSVC)
    target_compile_definitions(${t} PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
    target_compile_options(${t} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/O2 /EHsc>)
  endif()
endforeach()
target_compile_definitions(lowpoly-test PRIVATE LOWPOLY_TESTS)

enable_testing()
add_test(NAME lowpoly COMMAND lowpoly-test)
